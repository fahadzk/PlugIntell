/** Console interaction is enabled only by the CLI entry point, never by the web host. */
final class CliSupport {
  static boolean enabled, cliMode, citations=true;
  static boolean includeCitations(){return cliMode?citations:Boolean.parseBoolean(System.getenv().getOrDefault("RAG_SEARCH_INCLUDE_CITATIONS","true"));}
  private static final Map<String,String> values=new HashMap<>();
  private static final BufferedReader input=new BufferedReader(new InputStreamReader(System.in));
  static String ask(String question,boolean sensitive,String fallback){
    if(!enabled)throw new IllegalArgumentException(question+" Supply the setting before running, or enable interactive CLI input.");
    try{
      String value;var console=System.console();
      if(console!=null&&sensitive){char[] chars=console.readPassword("%s",question);if(chars==null)throw new IllegalArgumentException("Input ended.");value=new String(chars);Arrays.fill(chars,'\0');}
      else {if(sensitive&&console==null)System.err.println("No secure console: input may be echoed. Use an environment variable to hide it.");System.out.print(question);System.out.flush();value=console!=null?console.readLine():input.readLine();}
      if(value==null)throw new IllegalArgumentException("Input ended. Configure values in the environment for unattended runs.");
      if(value.isBlank()){if(fallback!=null)return fallback;throw new IllegalArgumentException("A value is required.");}return value;
    }catch(IOException e){throw new IllegalStateException("Cannot read console input.");}
  }
  static String environment(String name,boolean sensitive){
    String value=values.getOrDefault(name,System.getenv(name));
    if(value==null||value.isBlank()){value=ask(name+" is not set. Enter a value for this run: ",sensitive,null);values.put(name,value);}
    if(sensitive)RuntimeSupport.register(value);
    return value;
  }
  static Map<String,Object> settings(Map<String,Object> source,Map<String,String> required){
    Map<String,Object> result=new LinkedHashMap<>(source);
    for(var field:required.entrySet()){
      Object current=result.get(field.getKey());
      if(current!=null&&!current.toString().isBlank())continue;
      String value=environment(field.getKey(),field.getKey().matches("(?i).*(PASSWORD|SECRET|TOKEN|KEY|CREDENTIAL).*"));
      Object typed=switch(field.getValue()){case "integer"->Long.valueOf(value);case "number"->Double.valueOf(value);case "boolean"->{if(!Set.of("true","false").contains(value.toLowerCase(Locale.ROOT)))throw new IllegalArgumentException("Expected true or false for "+field.getKey());yield Boolean.valueOf(value);}default->value;};
      result.put(field.getKey(),typed);
    }
    return result;
  }
  static Map<String,Object> text2CypherSettings(Map<String,Object> source,Map<String,String> required){
    Map<String,Object> result=new LinkedHashMap<>(source);Map<String,String> needed=new LinkedHashMap<>(required);
    java.util.function.BiFunction<String,String,String> configured=(name,fallback)->Objects.toString(source.getOrDefault(name,System.getenv().getOrDefault(name,fallback)),"").strip();
    String provider=configured.apply("LLM_PROVIDER","OLLAMA").toUpperCase(Locale.ROOT);
    if(enabled&&cliMode){
      while(true){
        String selected=ask("Chat provider [OLLAMA/OPENAI/GEMINI] (Enter for "+provider+"): ",false,provider).strip().toUpperCase(Locale.ROOT);
        if(Set.of("OLLAMA","OPENAI","GEMINI").contains(selected)){provider=selected;break;}
        System.out.println("Choose OLLAMA, OPENAI or GEMINI.");
      }
    }
    if(!Set.of("OLLAMA","OPENAI","GEMINI").contains(provider))throw new IllegalArgumentException("LLM_PROVIDER must be OLLAMA, OPENAI or GEMINI.");
    result.put("LLM_PROVIDER",provider);
    for(String other:List.of("OLLAMA","OPENAI","GEMINI")){
      String key=other.equals("OLLAMA")?"OLLAMA_CHAT_MODEL":other+"_MODEL";needed.remove(key);needed.remove(other+"_API_KEY");
      if(!other.equals(provider))for(String field:new ArrayList<>(result.keySet()))if(field.startsWith(other+"_"))result.remove(field);
    }
    String modelKey=provider.equals("OLLAMA")?"OLLAMA_CHAT_MODEL":provider+"_MODEL",model=configured.apply(modelKey,"");
    if(enabled&&cliMode){
      while(true){String answer=ask("Chat model"+(model.isBlank()?"": " (Enter for "+model+")")+": ",false,model).strip();if(!answer.isBlank()){model=answer;break;}System.out.println("A chat model is required.");}
      String baseKey=provider+"_BASE_URL",base=configured.apply(baseKey,provider.equals("OLLAMA")?"http://localhost:11434":provider.equals("OPENAI")?"https://api.openai.com/v1/":"https://generativelanguage.googleapis.com");
      result.put(baseKey,ask(baseKey+" (Enter for "+base+"): ",false,base).strip());
    }
    if(!model.isBlank())result.put(modelKey,model);needed.put(modelKey,"string");
    if(!provider.equals("OLLAMA")){
      String key=provider+"_API_KEY";
      // Resolve through the existing secret path so secure console handling/redaction remains shared.
      if(!result.containsKey(key))result.put(key,"secret://env/"+key);
    }
    return settings(result,needed);
  }
  static Map<String,Object> ontologySettings(String pluginId,Map<String,Object> source){
    if(!pluginId.equals("gliner-graph-extraction"))return source;
    java.util.function.Function<String,String> configured=name->Objects.toString(source.getOrDefault(name,System.getenv().getOrDefault(name,"")),"").strip();
    Map<String,Object> result=new LinkedHashMap<>(source);
    String flag=configured.apply("GRAPH_GENERATE_ONTOLOGY");
    if(!flag.isEmpty()&&!flag.equalsIgnoreCase("true")&&!flag.equalsIgnoreCase("false"))throw new IllegalArgumentException("GRAPH_GENERATE_ONTOLOGY must be true or false.");
    boolean generate=flag.isEmpty()||Boolean.parseBoolean(flag);
    boolean hasSchema=!configured.apply("GRAPH_SCHEMA_JSON").isEmpty(),hasLlm=!configured.apply("GRAPH_LLM_URL").isEmpty();
    if(enabled&&cliMode&&!hasSchema&&hasLlm){
      while(true){
        String answer=ask("Generate ontology using the LLM? This sends a potentially billable request. "+(generate?"[Y/n]: ":"[y/N]: "),false,generate?"yes":"no").strip().toLowerCase(Locale.ROOT);
        if(Set.of("y","yes","true","n","no","false").contains(answer)){generate=Set.of("y","yes","true").contains(answer);break;}
        System.out.println("Enter y or n.");
      }
    }
    result.put("GRAPH_GENERATE_ONTOLOGY",generate);
    if(!generate||hasSchema){
      result.remove("GRAPH_LLM_API_KEY");
      if(!generate&&!hasSchema&&enabled&&cliMode&&yesNo("Want to express ontology manually [Y/n]? ",true))result.put("GRAPH_SCHEMA_JSON",manualOntology());
      return result;
    }
    if(!enabled||!cliMode||!hasLlm||!configured.apply("GRAPH_PROMPT").isEmpty())return result;
    result.put("GRAPH_PROMPT",ask("Ontology instructions (Enter to use the default compact ontology prompt): ",false,""));
    return result;
  }
  private static boolean yesNo(String prompt,boolean fallback){
    while(true){
      String answer=ask(prompt,false,fallback?"yes":"no").strip().toLowerCase(Locale.ROOT);
      if(Set.of("y","yes","true").contains(answer))return true;
      if(Set.of("n","no","false").contains(answer))return false;
      System.out.println("Enter y or n.");
    }
  }
  private static String ontologyNameKey(String name){return name.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9]","");}
  private static String manualOntology(){
    Map<String,String> names=new LinkedHashMap<>();
    while(true){
      String input=ask("Enter Ontology entity types: ",false,"");
      try{
        names.clear();String[] parts=input.split(",",-1);
        if(parts.length>20)throw new IllegalArgumentException("Maximum 20 entity types.");
        for(String part:parts){String name=part.strip();
          if(!name.matches("[A-Za-z][A-Za-z0-9_]{0,63}"))throw new IllegalArgumentException("Use comma-separated entity names, e.g. Train,Station (letters, digits and underscores; start with a letter; maximum 64 characters).");
          if(names.putIfAbsent(ontologyNameKey(name),name)!=null)throw new IllegalArgumentException("Entity types must be unique.");
        }
        break;
      }catch(IllegalArgumentException invalid){System.out.println(invalid.getMessage());}
    }
    List<Map<String,Object>> relationships=new ArrayList<>();
    while(true){
      String input=ask("Enter Ontology relationship types: ",false,"");
      try{
        relationships.clear();Set<String> seen=new HashSet<>();
        if(input.isBlank())break;
        String[] parts=input.split(",",-1);if(parts.length>30)throw new IllegalArgumentException("Maximum 30 relationship types.");
        for(String part:parts){
          var match=java.util.regex.Pattern.compile("\\s*([A-Z][A-Z0-9_]{0,63})\\s*\\(\\s*([A-Za-z][A-Za-z0-9_]{0,63})\\s*->\\s*([A-Za-z][A-Za-z0-9_]{0,63})\\s*\\)\\s*").matcher(part);
          if(!match.matches())throw new IllegalArgumentException("Use comma-separated relationships, e.g. STOPS_AT (Train -> Station), NEXT_STOP (Station -> Station). Use UPPER_SNAKE_CASE names, or Enter for none.");
          String name=match.group(1),head=names.get(ontologyNameKey(match.group(2))),tail=names.get(ontologyNameKey(match.group(3)));
          if(head==null||tail==null)throw new IllegalArgumentException("Relationship endpoints must reference declared entity types: "+String.join(", ",names.values())+".");
          if(!seen.add(ontologyNameKey(name)))throw new IllegalArgumentException("Relationship types must be unique.");
          relationships.add(Map.of("name",name,"description","","head",head,"tail",tail,"properties",List.of()));
        }
        break;
      }catch(IllegalArgumentException invalid){System.out.println(invalid.getMessage());}
    }
    List<Map<String,Object>> entities=new ArrayList<>();for(String name:names.values())entities.add(Map.of("name",name,"description","","properties",List.of("name")));
    return EventJson.write(Map.of("domain","Manual","entities",entities,"relationships",relationships));
  }
  static boolean missingSetting(RuntimeException error,Map<String,Object> config){
    if(!enabled)return false;
    for(Throwable cause=error;cause!=null;cause=cause.getCause()){
      var match=java.util.regex.Pattern.compile("\\b([A-Z][A-Z0-9_]+) is required\\b").matcher(Objects.toString(cause.getMessage(),""));
      if(match.find()){
        String name=match.group(1);if(config.containsKey(name)&&!Objects.toString(config.get(name),"").isBlank())return false;
        config.put(name,environment(name,name.matches(".*(PASSWORD|SECRET|TOKEN|KEY|CREDENTIAL).*")));return true;
      }
    }
    return false;
  }
  static void citations(String option){
    String configured=option==null?System.getenv("RAG_SEARCH_INCLUDE_CITATIONS"):option;
    if(configured==null||configured.isBlank())configured=enabled?ask("Display source citations? [Y/n]: ",false,"yes"):"yes";
    citations=switch(configured.toLowerCase(Locale.ROOT)){case "y","yes","true"->true;case "n","no","false"->false;default->throw new IllegalArgumentException("Citations must be yes or no.");};
  }
  static EventAdapter consoleEvents(){return event->{
    String viewer=viewerUrl(event);if(viewer!=null){System.out.println("Response viewer: "+viewer+" (Ctrl+C stops this application)");return;}
    String plugin=Objects.toString(event.pluginId(),"pipeline");
    switch(event.type()){
      case "NODE_STARTED"->System.out.println("\n["+plugin+"] Running");
      case "NODE_SUCCEEDED"->System.out.println("["+plugin+"] Finished | "+summary(event.payload().get("outputSummary"))+" | "+event.durationMillis()+" ms");
      case "NODE_FAILED"->System.err.println("["+plugin+"] FAILED: "+event.message());
      case "PROGRESS"->System.out.println("["+plugin+"] "+event.payload().get("completed")+" / "+event.payload().get("total")+" "+event.payload().getOrDefault("unit",""));
      case "PLUGIN_MESSAGE"->System.out.println("["+plugin+"] "+event.severity()+": "+event.message());
      default->{}
    }
  };}
  // Reconstruct only a fixed loopback URL from typed numeric fields, never from untrusted message text.
  // This avoids redacting public UUID/host digits when a short secret happens to contain those digits.
  private static String viewerUrl(ExecutionEvent event){
    if(!"universal-response".equals(event.pluginId())||!"response.viewer.available".equals(event.code()))return null;
    if(!(event.payload().get("viewerPort") instanceof Number port)||port.intValue()<1||port.intValue()>65535)return null;
    if(!(event.payload().get("viewerIdBytes") instanceof List<?> parts)||parts.size()!=16)return null;
    byte[] bytes=new byte[16];for(int i=0;i<16;i++){if(!(parts.get(i) instanceof Number n)||n.intValue()<-128||n.intValue()>127||n.doubleValue()!=n.intValue())return null;bytes[i]=(byte)n.intValue();}
    var buffer=java.nio.ByteBuffer.wrap(bytes);return "http://127.0.0.1:"+port.intValue()+"/responses/"+new UUID(buffer.getLong(),buffer.getLong());
  }
  private static String summary(Object value){if(!(value instanceof Map<?,?> fields))return "";return String.join(" | ",fields.entrySet().stream().map(e->e.getKey()+": "+e.getValue()).toList());}
  static int display(Object result){
    if(result instanceof GraphImportResult imported){System.out.println("\nNeo4j file import committed");System.out.println("  File: "+imported.fileName()+" | Format: "+imported.format()+" | Database: "+imported.database());System.out.println("  Created: "+imported.nodesCreated()+" nodes, "+imported.relationshipsCreated()+" relationships");System.out.println("  Deleted: "+imported.nodesDeleted()+" nodes, "+imported.relationshipsDeleted()+" relationships | Properties set: "+imported.propertiesSet());System.out.println("  Statements executed: "+imported.statementsExecuted());return 0;}
    if(result instanceof GraphWriteResult graph){System.out.println("\nGraph committed to Neo4j");System.out.println("  Database: "+graph.database()+" | Namespace: "+graph.namespace());System.out.println("  Entities: "+graph.entitiesStored()+" | Relationships: "+graph.relationshipsStored()+" | Source chunks: "+graph.chunksStored());return 0;}
    if(result instanceof ExtractedGraph graph){
      var metadata=graph.metadata();boolean complete=Boolean.TRUE.equals(metadata.get("complete"));
      System.out.println("\nGraph extraction "+(complete?"complete":"INCOMPLETE"));
      System.out.println("  Chunks: "+graph.chunks().size()+" | Failed: "+metadata.getOrDefault("chunks_failed",0)+" | Infeasible: "+metadata.getOrDefault("chunks_infeasible",0));
      System.out.println("  Entities: "+graph.entities().size()+" | Relationships: "+graph.relationships().size());
      System.out.println("  Ontology: "+graph.schema().get("domain")+" ("+graph.schema().get("source")+")");
      if(metadata.get("failure_details") instanceof List<?> failures&&!failures.isEmpty()){
        System.out.println("  Failures (up to 10):");
        for(Object item:failures.stream().limit(10).toList())if(item instanceof Map<?,?> failure){String id=Objects.toString(failure.get("chunk_id"),"");String source=graph.chunks().stream().filter(c->c.id().equals(id)).findFirst().map(c->c.sourceName()+", page "+c.pageNumber()+", chunk "+(c.ordinal()+1)).orElse(id);System.out.println("    "+source+": "+failure.get("reason"));}
      }
      System.out.println("\nEntities (up to 20)");
      for(var entity:graph.entities().stream().limit(20).toList()){System.out.println("  "+entity.get("type")+": "+entity.get("name"));if(citations)System.out.println("    Sources: "+entity.get("chunk_ids"));}
      Map<Object,String> names=new HashMap<>();graph.entities().forEach(e->names.put(e.get("id"),Objects.toString(e.get("name"))));
      System.out.println("Relationships (up to 20)");
      for(var edge:graph.relationships().stream().limit(20).toList())System.out.println("  "+names.get(edge.get("source_id"))+" --"+edge.get("type")+"--> "+names.get(edge.get("target_id")));
      if(citations&&!graph.entities().isEmpty()){System.out.println("Source index");for(var chunk:graph.chunks())System.out.println("  "+chunk.id()+": "+chunk.sourceName()+", page "+chunk.pageNumber());}
      return complete?0:2;
    }
    if(result instanceof AnswerResult answer){System.out.println("\nAnswer\n"+answer.answer());if(citations)for(var citation:answer.citations())System.out.println("  "+citation.sourceName()+", page "+citation.pageNumber()+" ("+citation.id()+")");return 0;}
    if(result instanceof String text)System.out.println("\nResult\n"+text);
    System.out.println("\nPipeline completed. "+DataSummaries.of(result));return 0;
  }
